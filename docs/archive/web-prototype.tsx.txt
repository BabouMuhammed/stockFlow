function QueryState({ error, isLoading, children }: { error: Error | null; isLoading: boolean; children: ReactNode }) {
  if (isLoading) return <div className="state-panel"><LoaderCircle className="spin" size={22} />Loading data…</div>;
  if (error) return <div className="state-panel error-state"><CircleAlert size={20} /><div><strong>Could not load data</strong><p>{error.message}. Check that the API and PostgreSQL are running.</p></div></div>;
  return <>{children}</>;
}

const productSchema = z.object({ name: z.string().trim().min(1, 'Enter a product name'), sku: z.string().trim().min(1, 'Enter a SKU'), barcode: z.string().optional(), categoryId: z.string().optional(), purchaseCost: z.coerce.number().min(0), sellingPrice: z.coerce.number().positive(), lowStockThreshold: z.coerce.number().int().min(0), isActive: z.boolean() });
type ProductFormValues = z.infer<typeof productSchema>;
function ProductDialog({ product, categories, onClose }: { product?: Product; categories: Category[]; onClose: () => void }) {
  const client = useQueryClient();
  const form = useForm<ProductFormValues>({ resolver: zodResolver(productSchema), defaultValues: product ? { name: product.name, sku: product.sku, barcode: product.barcode ?? '', categoryId: product.category?.id ?? '', purchaseCost: product.purchaseCostMinor / 100, sellingPrice: product.sellingPriceMinor / 100, lowStockThreshold: product.lowStockThreshold, isActive: product.isActive } : { name: '', sku: '', barcode: '', categoryId: '', purchaseCost: 0, sellingPrice: 0, lowStockThreshold: 5, isActive: true } });
  const mutation = useMutation({ mutationFn: (values: ProductFormValues) => request<Product>(product ? `/products/${product.id}` : '/products', { method: product ? 'PATCH' : 'POST', body: JSON.stringify({ name: values.name, sku: values.sku, barcode: values.barcode || null, categoryId: values.categoryId || null, purchaseCostMinor: Math.round(values.purchaseCost * 100), sellingPriceMinor: Math.round(values.sellingPrice * 100), lowStockThreshold: values.lowStockThreshold, isActive: values.isActive }) }), onSuccess: async () => { await client.invalidateQueries({ queryKey: ['products'] }); onClose(); } });
  return <Modal title={product ? 'Edit product' : 'Add product'} onClose={onClose}><form className="form-grid" onSubmit={form.handleSubmit((values) => mutation.mutate(values))}>
    <label className="field full-field">Product name<input {...form.register('name')} placeholder="e.g. Rice 25kg" />{form.formState.errors.name && <small className="field-error">{form.formState.errors.name.message}</small>}</label>
    <label className="field">SKU<input {...form.register('sku')} placeholder="RICE-25" />{form.formState.errors.sku && <small className="field-error">{form.formState.errors.sku.message}</small>}</label>
    <label className="field">Barcode<input {...form.register('barcode')} placeholder="Optional" /></label>
    <label className="field">Category<select {...form.register('categoryId')}><option value="">Uncategorised</option>{categories.map((category) => <option value={category.id} key={category.id}>{category.name}</option>)}</select></label>
    <label className="field">Purchase cost (GMD)<input type="number" step="0.01" min="0" {...form.register('purchaseCost')} /></label>
    <label className="field">Selling price (GMD)<input type="number" step="0.01" min="0.01" {...form.register('sellingPrice')} />{form.formState.errors.sellingPrice && <small className="field-error">{form.formState.errors.sellingPrice.message}</small>}</label>
    <label className="field">Low-stock threshold<input type="number" min="0" step="1" {...form.register('lowStockThreshold')} /></label>
    <label className="check-field full-field"><input type="checkbox" {...form.register('isActive')} /> Active and available for sale</label>
    {mutation.error && <div className="form-alert full-field">{mutation.error.message}</div>}
    <div className="modal-actions full-field"><button type="button" className="button secondary" onClick={onClose}>Cancel</button><button className="button primary" disabled={mutation.isPending}>{mutation.isPending ? 'Saving…' : product ? 'Save changes' : 'Add product'}</button></div>
  </form></Modal>;
}

function PosPage() {
  const client = useQueryClient(); const [search, setSearch] = useState(''); const [cart, setCart] = useState<CartLine[]>([]); const [paymentMethod, setPaymentMethod] = useState<'CASH' | 'MOBILE_MONEY'>('CASH'); const [receipt, setReceipt] = useState<Sale | null>(null); const [message, setMessage] = useState('');
  const productsQuery = useQuery({ queryKey: ['pos-products'], queryFn: () => request<{ data: Product[] }>('/products?active=true&limit=100') });
  const products = productsQuery.data?.data ?? [];
  const filtered = products.filter((p) => `${p.name} ${p.sku} ${p.barcode ?? ''}`.toLowerCase().includes(search.toLowerCase())).slice(0, 20);
  const total = cart.reduce((sum, line) => sum + line.product.sellingPriceMinor * line.quantity, 0);
  const add = (product: Product) => { setMessage(''); setCart((current) => { const existing = current.find((line) => line.product.id === product.id); const nextQuantity = (existing?.quantity ?? 0) + 1; if (nextQuantity > (product.stockLevel?.quantity ?? 0)) { setMessage(`Only ${product.stockLevel?.quantity ?? 0} ${product.name} available.`); return current; } return existing ? current.map((line) => line.product.id === product.id ? { ...line, quantity: nextQuantity } : line) : [...current, { product, quantity: 1 }]; }); };
  const checkout = useMutation({ mutationFn: () => request<Sale>('/sales', { method: 'POST', headers: { 'Idempotency-Key': newKey() }, body: JSON.stringify({ paymentMethod, items: cart.map((line) => ({ productId: line.product.id, quantity: line.quantity })) }) }), onSuccess: async (sale) => { setReceipt(sale); setCart([]); setSearch(''); setMessage(''); await Promise.all([client.invalidateQueries({ queryKey: ['pos-products'] }), client.invalidateQueries({ queryKey: ['products'] }), client.invalidateQueries({ queryKey: ['inventory'] }), client.invalidateQueries({ queryKey: ['movements'] }), client.invalidateQueries({ queryKey: ['summary'] }), client.invalidateQueries({ queryKey: ['sales'] }), client.invalidateQueries({ queryKey: ['reports'] })]); }, onError: (error) => setMessage(error.message) });
  const onSearchKeyDown = (event: KeyboardEvent<HTMLInputElement>) => { if (event.key === 'Enter' && search.trim()) { const exact = products.find((p) => p.sku.toLowerCase() === search.toLowerCase() || p.barcode === search.trim()); if (exact) { add(exact); setSearch(''); } } };
  return <><div className="pos-toolbar"><div className="pos-search"><Search size={18} /><input autoFocus value={search} onChange={(event) => setSearch(event.target.value)} onKeyDown={onSearchKeyDown} placeholder="Search product, SKU or scan barcode" /><kbd>Enter</kbd></div><span className="scanner-hint"><span className="status-dot" /> Scanner ready</span></div><QueryState error={productsQuery.error} isLoading={productsQuery.isLoading}><div className="pos-layout"><section className="pos-products"><div className="pos-section-heading"><div><h2>Products</h2><p>{filtered.length} available items</p></div><span className="category-chip"><Boxes size={14} /> All products</span></div>{filtered.length ? <div className="pos-product-grid">{filtered.map((product) => { const stock = product.stockLevel?.quantity ?? 0; return <button className={`pos-product ${stock === 0 ? 'sold-out' : ''}`} key={product.id} onClick={() => add(product)} disabled={stock === 0}><span className="pos-product-tag">{product.category?.name ?? 'General'}</span><span className="pos-product-name">{product.name}</span><span className="pos-product-sku">{product.sku}</span><span className="pos-product-bottom"><strong>{money(product.sellingPriceMinor)}</strong><small className={stock <= product.lowStockThreshold ? 'warning-text' : ''}>{stock} available</small></span><span className="add-product"><Plus size={16} /></span></button>; })}</div> : <EmptyState title="No matching products" detail="Adjust your search or scan another barcode." />}</section>
    <aside className="cart-panel"><div className="cart-heading"><div className="cart-icon"><ShoppingBasket size={18} /></div><div><h2>Current sale</h2><p>{cart.length} {cart.length === 1 ? 'product' : 'products'}</p></div>{cart.length > 0 && <button className="quiet-action clear-cart" onClick={() => { setCart([]); setMessage(''); }}>Clear</button>}</div><div className="cart-lines">{cart.length ? cart.map((line) => <div className="cart-line" key={line.product.id}><div className="cart-line-main"><strong>{line.product.name}</strong><span>{money(line.product.sellingPriceMinor)} each</span></div><div className="cart-line-controls"><button aria-label={`Decrease ${line.product.name}`} onClick={() => setCart((current) => current.flatMap((item) => item.product.id !== line.product.id ? [item] : item.quantity > 1 ? [{ ...item, quantity: item.quantity - 1 }] : []))}>−</button><span>{line.quantity}</span><button aria-label={`Increase ${line.product.name}`} onClick={() => add(line.product)}>+</button><button className="remove-line" aria-label={`Remove ${line.product.name}`} onClick={() => setCart((current) => current.filter((item) => item.product.id !== line.product.id))}><Trash2 size={15} /></button></div><strong className="cart-line-total">{money(line.product.sellingPriceMinor * line.quantity)}</strong></div>) : <div className="cart-empty"><div><ShoppingCart size={23} /></div><strong>Your cart is ready</strong><span>Add a product to start a sale.</span></div>}</div><div className="cart-checkout"><div className="payment-title">Payment method</div><div className="payment-options"><button className={paymentMethod === 'CASH' ? 'payment-option selected' : 'payment-option'} onClick={() => setPaymentMethod('CASH')}><span className="payment-radio" /><span><strong>Cash</strong><small>Cash payment</small></span><CreditCard size={17} /></button><button className={paymentMethod === 'MOBILE_MONEY' ? 'payment-option selected' : 'payment-option'} onClick={() => setPaymentMethod('MOBILE_MONEY')}><span className="payment-radio" /><span><strong>Mobile money</strong><small>Recorded only</small></span><Activity size={17} /></button></div><div className="total-row"><span>Total due</span><strong>{money(total)}</strong></div><p className="mobile-money-note">Mobile money is recorded manually; no transfer is processed.</p>{message && <div className="form-alert compact-alert">{message}</div>}{checkout.error && !message && <div className="form-alert compact-alert">{checkout.error.message}</div>}<button className="button primary checkout-button" disabled={!cart.length || checkout.isPending} onClick={() => checkout.mutate()}>{checkout.isPending ? <LoaderCircle className="spin" size={18} /> : <ShoppingCart size={18} />}{checkout.isPending ? 'Completing sale…' : 'Complete sale'}<span>{money(total)}</span></button></div></aside></div></QueryState>{receipt && <ReceiptDialog sale={receipt} onClose={() => setReceipt(null)} />}</>;
}
